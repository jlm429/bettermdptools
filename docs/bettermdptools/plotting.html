Preparation functions are pure and independent of Matplotlib. Renderers draw only on explicit caller-provided matplotlib.axes.Axes objects and return those axes. They never show, save, close, resize, or apply figure-level layout. Callers own the figure and its lifecycle.

117def aggregate_values(
118    values: ArrayLike,
119    shape: Sequence[int],
120    axes: Sequence[int],
121) -> NDArray[np.floating]:
122    """Average numeric values over axes from the original source shape.
123
124    Negative axes are supported. Duplicate axes are rejected, and an empty
125    axis sequence returns a copy of the reshaped input.
126    """
127    try:
128        source_shape = tuple(index(dimension) for dimension in shape)
129    except TypeError as error:
130        raise TypeError("shape must contain integers") from error
131    if not source_shape or any(dimension <= 0 for dimension in source_shape):
132        raise ValueError("shape must contain positive integers")
133    data = _numeric_array(values, "values")
134    if data.size != int(np.prod(source_shape)):
135        raise ValueError(
136            f"shape {source_shape} requires {int(np.prod(source_shape))} values, "
137            f"but received {data.size}"
138        )
139    data = data.reshape(source_shape)
140
141    normalized_axes: list[int] = []
142    for raw_axis in axes:
143        axis = index(raw_axis)
144        if axis < 0:
145            axis += data.ndim
146        if not 0 <= axis < data.ndim:
147            raise AxisError(axis, ndim=data.ndim)
148        normalized_axes.append(axis)
149    if len(set(normalized_axes)) != len(normalized_axes):
150        raise ValueError("axes must not contain duplicates")
151    if not normalized_axes:
152        return data.copy()
153    return np.mean(data, axis=tuple(normalized_axes))

def prepare_convergence( value_history: ArrayLike, *, policy_history: ArrayLike | None = None, valid_length: int | None = None, value_statistic: Literal['max', 'mean'] = 'max') -> ConvergenceData: 
262def prepare_convergence(
263    value_history: ArrayLike,
264    *,
265    policy_history: ArrayLike | None = None,
266    valid_length: int | None = None,
267    value_statistic: Literal["max", "mean"] = "max",
268) -> ConvergenceData:
269    """Prepare value deltas and policy-change counts between history entries.
270
271    The first dimension is time and every remaining dimension is flattened for
272    comparison. ``valid_length`` is the sole validity control. When omitted,
273    every supplied row is treated as valid. No zero-valued row is ever trimmed
274    or interpreted as padding.
275    """
276    values = _numeric_array(value_history, "value_history")
277    if values.ndim < 2:
278        raise ValueError("value_history must have time plus at least one value axis")
279    if np.isnan(values).any():
280        raise ValueError("value_history must not contain NaN values")
281    if valid_length is None:
282        length = values.shape[0]
283    else:
284        if (
285            isinstance(valid_length, bool)
286            or not isinstance(valid_length, Integral)
287            or not 2 <= valid_length <= values.shape[0]
288        ):
289            raise ValueError(
290                "valid_length must be an integer from 2 through the history length"
291            )
292        length = int(valid_length)
293    if length < 2:
294        raise ValueError("value_history must contain at least two valid entries")
295    if value_statistic not in {"max", "mean"}:
296        raise ValueError("value_statistic must be 'max' or 'mean'")
297
298    differences = np.abs(np.diff(values[:length], axis=0)).reshape(length - 1, -1)
299    reducer = np.max if value_statistic == "max" else np.mean
300    value_delta = reducer(differences, axis=1)
301
302    policy_changes = None
303    if policy_history is not None:
304        policies = np.asarray(policy_history)
305        if policies.ndim < 2:
306            raise ValueError(
307                "policy_history must have time plus at least one policy axis"
308            )
309        if policies.shape[0] < length:
310            raise ValueError("policy_history is shorter than valid_length")
311        flattened = policies[:length].reshape(length, -1)
312        policy_changes = np.count_nonzero(flattened[1:] != flattened[:-1], axis=1)
313

def prepare_policy_grid( policy: Mapping[int, int] | Sequence[int], values: ArrayLike, action_labels: Mapping[int, object] | Sequence[object] | None, shape: Sequence[int]) -> PolicyGridData: 
156def prepare_policy_grid(
157    policy: Mapping[int, int] | Sequence[int],
158    values: ArrayLike,
159    action_labels: Mapping[int, object] | Sequence[object] | None,
160    shape: Sequence[int],
161) -> PolicyGridData:
162    """Prepare a complete state-value and action-label policy grid.
163
164    The policy must define every state from zero through ``values.size - 1``.
165    Actions must be integral indices. When ``action_labels`` is omitted, the
166    numeric actions are converted to strings.
167    """
168    value_data = prepare_value_grid(values, shape)
169    flat_values = value_data.values.ravel()
170    labels: list[str] = []
171    for state in range(flat_values.size):
172        try:
173            action = policy[state]
174        except (IndexError, KeyError) as error:
175            raise ValueError(f"policy is missing state {state}") from error
176        if isinstance(action, bool) or not isinstance(action, Integral):
177            raise TypeError(f"policy action for state {state} must be an integer")
178        action = int(action)
179        if action < 0:
180            raise ValueError(f"policy action for state {state} must be non-negative")
181        if action_labels is None:
182            label = action
183        else:
184            try:
185                label = action_labels[action]
186            except (IndexError, KeyError) as error:
187                raise ValueError(f"action_labels is missing action {action}") from error
188        labels.append(str(label))
189    actions = np.asarray(labels, dtype=object).reshape(value_data.values.shape)
190    return PolicyGridData(values=value_data.values, actions=actions)
