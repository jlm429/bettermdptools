Reinforcement learning algorithms. RL expects a Gymnasium environment.

Model-free learning algorithms: Q-Learning and SARSA work out of the box with any gymnasium environments that have single discrete valued state spaces, like frozen lake. A lambda function is required to convert state spaces not in this format.

 50    @staticmethod
 51    def decay_schedule(
 52        init_value, min_value, decay_ratio, max_steps, log_start=-2, log_base=10
 53    ):
 54        """
 55        Generates a decay schedule for a given initial value.
 56
 57        Parameters
 58        ----------
 59        init_value : float
 60            Initial value of the quantity being decayed.
 61        min_value : float
 62            Minimum value init_value is allowed to decay to.
 63        decay_ratio : float
 64            The exponential factor exp(decay_ratio).
 65        max_steps : int
 66            Max iteration steps for decaying init_value.
 67        log_start : float, optional
 68            Starting value of the decay sequence, by default -2.
 69        log_base : float, optional
 70            Base of the log space, by default 10.
 71
 72        Returns
 73        -------
 74        np.ndarray
 75            Decay values where values[i] is the value used at i-th step.
 76        """
 77        decay_steps = int(max_steps * decay_ratio)
 78        rem_steps = max_steps - decay_steps
 79        values = np.logspace(log_start, 0, decay_steps, base=log_base, endpoint=True)[
 80            ::-1
 81        ]
 82        values = (values - values.min()) / (values.max() - values.min())
 83        values = (init_value - min_value) * values + min_value
 84        values = np.pad(values, (0, rem_steps), "edge")
 85        return values

 87    def q_learning(
 88        self,
 89        nS=None,
 90        nA=None,
 91        convert_state_obs=lambda state: state,
 92        gamma=0.99,
 93        init_alpha=0.5,
 94        min_alpha=0.01,
 95        alpha_decay_ratio=0.5,
 96        init_epsilon=1.0,
 97        min_epsilon=0.1,
 98        epsilon_decay_ratio=0.9,
 99        n_episodes=10000,
100    ):
101        """
102        Q-Learning algorithm.
103
104        Parameters
105        ----------
106        nS : int, optional
107            Number of states, by default None.
108        nA : int, optional
109            Number of available actions, by default None.
110        convert_state_obs : function, optional
111            Converts state into an integer, by default lambda state: state.
112        gamma : float, optional
113            Discount factor, by default 0.99.
114        init_alpha : float, optional
115            Initial learning rate, by default 0.5.
116        min_alpha : float, optional
117            Minimum learning rate, by default 0.01.
118        alpha_decay_ratio : float, optional
119            Decay schedule of learning rate for future iterations, by default 0.5.
120        init_epsilon : float, optional
121            Initial epsilon value for epsilon greedy strategy, by default 1.0.
122        min_epsilon : float, optional
123            Minimum epsilon, by default 0.1.
124        epsilon_decay_ratio : float, optional
125            Decay schedule of epsilon for future iterations, by default 0.9.
126        n_episodes : int, optional
127            Number of episodes for the agent, by default 10000.
128
129        Returns
130        -------
131        tuple
132            Q : np.ndarray
133                Final action-value function Q(s,a).
134            V : np.ndarray
135                State values array.
136            pi : dict
137                Policy mapping states to actions.
138            Q_track : np.ndarray
139                Log of Q(s,a) for each episode.
140            pi_track : list
141                Log of complete policy for each episode.
142            rewards : np.ndarray
143                Rewards obtained in each episode.
144
145        Notes
146        -----
147        Episodes stop after either termination or truncation. The TD target
148        bootstraps across truncation, but not across a true terminal state.
149        """
150        if nS is None:
151            nS = self.env.observation_space.n
152        if nA is None:
153            nA = self.env.action_space.n
154        pi_track = []
155        Q = np.zeros((nS, nA), dtype=np.float32)
156        Q_track = np.zeros((n_episodes, nS, nA), dtype=np.float32)
157        alphas = RL.decay_schedule(init_alpha, min_alpha, alpha_decay_ratio, n_episodes)
158        epsilons = RL.decay_schedule(
159            init_epsilon, min_epsilon, epsilon_decay_ratio, n_episodes
160        )
161        rewards = np.zeros(n_episodes, dtype=np.float32)
162        for e in tqdm(range(n_episodes), leave=False):
163            self.callbacks.on_episode_begin(self)
164            self.callbacks.on_episode(self, episode=e)
165            state, info = self.env.reset()
166            episode_done = False
167            state = convert_state_obs(state)
168            total_reward = 0
169            while not episode_done:
170                if self.render:
171                    warnings.warn(
172                        "Occasional rendering is deprecated. Use test_env.py "
173                        "to render."
174                    )
175                action = self.select_action(state, Q, epsilons[e])
176                next_state, reward, terminated, truncated, _ = self.env.step(action)
177                episode_done = terminated or truncated
178                self.callbacks.on_env_step(self)
179                next_state = convert_state_obs(next_state)
180                td_target = reward + gamma * Q[next_state].max() * (not terminated)
181                td_error = td_target - Q[state][action]
182                Q[state][action] = Q[state][action] + alphas[e] * td_error
183                state = next_state
184                total_reward += reward
185            rewards[e] = total_reward
186            Q_track[e] = Q
187            pi_track.append(np.argmax(Q, axis=1))
188            self.render = False
189            self.callbacks.on_episode_end(self)
190
191        V = np.max(Q, axis=1)
192
193        pi = {s: a for s, a in enumerate(np.argmax(Q, axis=1))}
194        return Q, V, pi, Q_track, pi_track, rewards

65            The exponential factor exp(decay_ratio).
66        max_steps : int
67            Max iteration steps for decaying init_value.
68        log_start : float, optional
69            Starting value of the decay sequence, by default -2.
70        log_base : float, optional
71            Base of the log space, by default 10.
72
73        Returns
74        -------
75        np.ndarray
76            Decay values where values[i] is the value used at i-th step.
77        """
78        decay_steps = int(max_steps * decay_ratio)
79        rem_steps = max_steps - decay_steps
80        values = np.logspace(log_start, 0, decay_steps, base=log_base, endpoint=True)[
81            ::-1
82        ]
83        values = (values - values.min()) / (values.max() - values.min())
84        values = (init_value - min_value) * values + min_value
85        values = np.pad(values, (0, rem_steps), "edge")
86        return values
