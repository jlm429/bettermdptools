140    def adaptive_angle_bins(self, angle_range, center_resolution, outer_resolution):
141        """
142        Generate adaptive angle bins with finer resolution near the center.
143
144        Parameters:
145        - angle_range (tuple): The minimum and maximum angles in radians.
146        - center_resolution (float): Bin width near zero angle for higher resolution.
147        - outer_resolution (float): Bin width away from zero for lower resolution.
148
149        Returns:
150        - np.array: An array of bin edges with adaptive spacing.
151        """
152        min_angle, max_angle = angle_range
153        # Generate finer bins around zero
154        center_bins = np.arange(
155            -center_resolution, center_resolution + 1e-6, center_resolution / 10
156        )
157        # Generate sparser bins outside the center region
158        left_bins = np.linspace(
159            min_angle,
160            -center_resolution,
161            num=int((-center_resolution - min_angle) / outer_resolution) + 1,
162            endpoint=False,
163        )
164        right_bins = np.linspace(
165            center_resolution,
166            max_angle,
167            num=np.max(
168                [int((max_angle - center_resolution) / outer_resolution) + 1, 2]
169            ),
170            endpoint=True,
171        )[1:]
172        return np.unique(np.concatenate([left_bins, center_bins, right_bins]))

188        """
189        Converts a single index into a multidimensional state representation.

207    def compute_next_state(
208        self, position_idx, velocity_idx, angle_idx, angular_velocity_idx, action
209    ):
210        """
211        Compute the next state from state indices and an action using simplified
212        physics.
213
214        Parameters:
215        - position_idx (int): Current index of the cart's position.
216        - velocity_idx (int): Current index of the cart's velocity.
217        - angle_idx (int): Current index of the pole's angle.
218        - angular_velocity_idx (int): Current index of the pole's angular velocity.
219        - action (int): Action taken (0 for left, 1 for right).
220
221        Returns:
222        - tuple: Next state index, reward, and episode termination flag.
223        """
224        position = np.linspace(*self.position_range, self.position_bins)[position_idx]
225        velocity = np.linspace(*self.velocity_range, self.velocity_bins)[velocity_idx]
226        angle = self.angle_bins[angle_idx]
227        angular_velocity = np.linspace(
228            *self.angular_velocity_range, self.angular_velocity_bins
229        )[angular_velocity_idx]
230
231        # Simulate physics here (simplified)
232        thresh = self.threshold_bins
233        force = 10 if action == 1 else -10
234        new_velocity = velocity + (force + np.sin(angle) * -10.0) * thresh
235        new_position = position + new_velocity * thresh
236        new_angular_velocity = angular_velocity + (3.0 * np.sin(angle) - force) * thresh
237        new_angle = angle + new_angular_velocity * thresh
238
239        new_position_idx = np.clip(
240            np.digitize(
241                new_position, np.linspace(*self.position_range, self.position_bins)
242            )
243            - 1,
244            0,
245            self.position_bins - 1,
246        )
247        new_velocity_idx = np.clip(
248            np.digitize(
249                new_velocity, np.linspace(*self.velocity_range, self.velocity_bins)
250            )
251            - 1,
252            0,
253            self.velocity_bins - 1,
254        )
255        new_angle_idx = np.clip(
256            np.digitize(new_angle, self.angle_bins) - 1, 0, len(self.angle_bins) - 1
257        )
258        new_angular_velocity_idx = np.clip(
259            np.digitize(
260                new_angular_velocity,
261                np.linspace(*self.angular_velocity_range, self.angular_velocity_bins),
262            )
263            - 1,
264            0,
265            self.angular_velocity_bins - 1,
266        )
267
268        new_state_idx = np.ravel_multi_index(
269            (
270                new_position_idx,
271                new_velocity_idx,
272                new_angle_idx,
273                new_angular_velocity_idx,
274            ),
275            (
276                self.position_bins,
277                self.velocity_bins,
278                len(self.angle_bins),
279                self.angular_velocity_bins,
280            ),
281        )
282
283        reward = 1 if np.abs(new_angle) < 12 * np.pi / 180 else -1
284        done = (
285            True
286            if np.abs(new_angle) >= 12 * np.pi / 180 or np.abs(new_position) > 2.4
287            else False
288        )
289
290        return new_state_idx, reward, done
