Reinforcement learning algorithms. RL expects a Gymnasium environment.

Model-free learning algorithms: Q-Learning and SARSA work out of the box with any gymnasium environments that have single discrete valued state spaces, like frozen lake. A lambda function is required to convert state spaces not in this format.

 71    @staticmethod
 72    def decay_schedule(
 73        init_value, min_value, decay_ratio, max_steps, log_start=-2, log_base=10
 74    ):
 75        """
 76        Generates a decay schedule for a given initial value.
 77
 78        Parameters
 79        ----------
 80        init_value : float
 81            Finite initial value of the quantity being decayed.
 82        min_value : float
 83            Finite minimum value, no greater than `init_value`.
 84        decay_ratio : float
 85            Fraction of `max_steps` over which values decay, in (0, 1].
 86        max_steps : int
 87            Positive number of steps in the returned schedule.
 88        log_start : float, optional
 89            Finite negative start of the logarithmic sequence, by default -2.
 90        log_base : float, optional
 91            Finite logarithm base greater than 1, by default 10.
 92
 93        Returns
 94        -------
 95        np.ndarray
 96            Decay values where values[i] is the value used at i-th step.
 97        """
 98        if (
 99            isinstance(max_steps, bool)
100            or not isinstance(max_steps, Integral)
101            or max_steps < 1
102        ):
103            raise ValueError("max_steps must be a positive integer")
104        if (
105            isinstance(decay_ratio, bool)
106            or not isinstance(decay_ratio, Real)
107            or not np.isfinite(decay_ratio)
108            or not 0 < decay_ratio <= 1
109        ):
110            raise ValueError("decay_ratio must be finite and in (0, 1]")
111        if any(
112            isinstance(value, bool)
113            or not isinstance(value, Real)
114            or not np.isfinite(value)
115            for value in (init_value, min_value)
116        ):
117            raise ValueError("init_value and min_value must be finite")
118        if init_value < min_value:
119            raise ValueError("init_value must be greater than or equal to min_value")
120        if (
121            isinstance(log_start, bool)
122            or not isinstance(log_start, Real)
123            or not np.isfinite(log_start)
124            or log_start >= 0
125        ):
126            raise ValueError("log_start must be finite and negative")
127        if (
128            isinstance(log_base, bool)
129            or not isinstance(log_base, Real)
130            or not np.isfinite(log_base)
131            or log_base <= 1
132        ):
133            raise ValueError("log_base must be finite and greater than 1")
134
135        max_steps = int(max_steps)
136        if max_steps == 1 or init_value == min_value:
137            return np.full(max_steps, init_value, dtype=float)
138
139        decay_steps = min(max_steps, max(2, int(max_steps * decay_ratio)))
140        rem_steps = max_steps - decay_steps
141        values = np.logspace(log_start, 0, decay_steps, base=log_base, endpoint=True)[
142            ::-1
143        ]
144        values = (values - values.min()) / (values.max() - values.min())
145        values = (init_value - min_value) * values + min_value
146        values = np.pad(values, (0, rem_steps), constant_values=min_value)
147        return values

149    def q_learning(
150        self,
151        nS=None,
152        nA=None,
153        convert_state_obs=lambda state: state,
154        gamma=0.99,
155        init_alpha=0.5,
156        min_alpha=0.01,
157        alpha_decay_ratio=0.5,
158        init_epsilon=1.0,
159        min_epsilon=0.1,
160        epsilon_decay_ratio=0.9,
161        n_episodes=10000,
162        seed=None,
163        callbacks: CallbackSpec | None = None,
164    ):
165        """
166        Q-Learning algorithm.
167
168        Parameters
169        ----------
170        nS : int, optional
171            Number of states, by default None.
172        nA : int, optional
173            Number of available actions, by default None.
174        convert_state_obs : function, optional
175            Converts state into an integer, by default lambda state: state.
176        gamma : float, optional
177            Discount factor, by default 0.99.
178        init_alpha : float, optional
179            Initial learning rate, by default 0.5.
180        min_alpha : float, optional
181            Minimum learning rate, by default 0.01.
182        alpha_decay_ratio : float, optional
183            Learning-rate schedule ratio passed to `decay_schedule`, by default 0.5.
184        init_epsilon : float, optional
185            Initial epsilon value for epsilon greedy strategy, by default 1.0.
186        min_epsilon : float, optional
187            Minimum epsilon, by default 0.1.
188        epsilon_decay_ratio : float, optional
189            Exploration schedule ratio passed to `decay_schedule`, by default 0.9.
190        n_episodes : int, optional
191            Positive number of episodes for the agent, by default 10000.
192        seed : int, optional
193            Seed passed to the first environment reset. Later resets continue
194            the environment's seeded random number sequence.
195        callbacks : Callbacks or iterable of Callbacks, optional
196            Callback object or ordered callback iterable for this run. When
197            omitted, callbacks supplied to ``RL`` or assigned to
198            ``rl.callbacks`` are used.
199
200        Returns
201        -------
202        tuple
203            Q : np.ndarray
204                Final action-value function Q(s,a).
205            V : np.ndarray
206                State values array.
207            pi : dict
208                Policy mapping states to actions.
209            Q_track : np.ndarray
210                Log of Q(s,a) for each episode.
211            pi_track : list
212                Log of complete policy for each episode.
213            rewards : np.ndarray
214                Rewards obtained in each episode.
215
216        Notes
217        -----
218        Episodes stop after either termination or truncation. The TD target
219        bootstraps across truncation, but not across a true terminal state.
220        """
221        if nS is None:
222            nS = self.env.observation_space.n
223        if nA is None:
224            nA = self.env.action_space.n
225        pi_track = []
226        Q = np.zeros((nS, nA), dtype=np.float32)
227        Q_track = np.zeros((n_episodes, nS, nA), dtype=np.float32)
228        alphas = RL.decay_schedule(init_alpha, min_alpha, alpha_decay_ratio, n_episodes)
229        epsilons = RL.decay_schedule(
230            init_epsilon, min_epsilon, epsilon_decay_ratio, n_episodes
231        )
232        rewards = np.zeros(n_episodes, dtype=np.float32)
233        active_callbacks = (
234            self.callbacks if callbacks is None else _normalize_callbacks(callbacks)
235        )
236        for e in tqdm(range(n_episodes), leave=False):
237            if e == 0 and seed is not None:
238                observation, info = self.env.reset(seed=seed)
239            else:
240                observation, info = self.env.reset()
241            state = convert_state_obs(observation)
242            if active_callbacks:
243                observation_snapshot = _snapshot_observation(observation)
244                start_context = EpisodeContext(
245                    algorithm="q_learning",
246                    episode=e,
247                    total_episodes=n_episodes,
248                    observation=observation_snapshot,
249                    state=state,
250                    info=_snapshot_info(info),
251                    steps=0,
252                    total_reward=0.0,
253                    terminated=False,
254                    truncated=False,
255                    epsilon=float(epsilons[e]),
256                    alpha=float(alphas[e]),
257                    gamma=float(gamma),
258                )
259                _dispatch_callbacks(
260                    active_callbacks, "on_episode_begin", self, start_context
261                )
262                _dispatch_callbacks(active_callbacks, "on_episode", self, start_context)
263            episode_done = False
264            total_reward = 0.0
265            steps = 0
266            terminated = False
267            truncated = False
268            while not episode_done:
269                if self.render:
270                    warnings.warn(
271                        "Occasional rendering is deprecated. Use test_env.py "
272                        "to render."
273                    )
274                action = self.select_action(state, Q, epsilons[e])
275                next_observation, reward, terminated, truncated, info = self.env.step(
276                    action
277                )
278                episode_done = terminated or truncated
279                next_state = convert_state_obs(next_observation)
280                steps += 1
281                total_reward += float(reward)
282                if active_callbacks:
283                    next_observation_snapshot = _snapshot_observation(next_observation)
284                    transition_context = TransitionContext(
285                        algorithm="q_learning",
286                        episode=e,
287                        total_episodes=n_episodes,
288                        step=steps,
289                        observation=observation_snapshot,
290                        state=state,
291                        action=action,
292                        reward=float(reward),
293                        terminated=terminated,
294                        truncated=truncated,
295                        next_observation=next_observation_snapshot,
296                        next_state=next_state,
297                        info=_snapshot_info(info),
298                        total_reward=total_reward,
299                        epsilon=float(epsilons[e]),
300                        alpha=float(alphas[e]),
301                        gamma=float(gamma),
302                    )
303                    _dispatch_callbacks(
304                        active_callbacks, "on_env_step", self, transition_context
305                    )
306                    observation_snapshot = next_observation_snapshot
307                td_target = reward + gamma * Q[next_state].max() * (not terminated)
308                td_error = td_target - Q[state][action]
309                Q[state][action] = Q[state][action] + alphas[e] * td_error
310                observation, state = next_observation, next_state
311            rewards[e] = total_reward
312            Q_track[e] = Q
313            pi_track.append(np.argmax(Q, axis=1))
314            self.render = False
315            if active_callbacks:
316                end_context = EpisodeContext(
317                    algorithm="q_learning",
318                    episode=e,
319                    total_episodes=n_episodes,
320                    observation=observation_snapshot,
321                    state=state,
322                    info=_snapshot_info(info),
323                    steps=steps,
324                    total_reward=total_reward,
325                    terminated=terminated,
326                    truncated=truncated,
327                    epsilon=float(epsilons[e]),
328                    alpha=float(alphas[e]),
329                    gamma=float(gamma),
330                )
331                _dispatch_callbacks(
332                    active_callbacks, "on_episode_end", self, end_context
333                )
334
335        V = np.max(Q, axis=1)
336
337        pi = {s: a for s, a in enumerate(np.argmax(Q, axis=1))}
338        return Q, V, pi, Q_track, pi_track, rewards

 74        init_value, min_value, decay_ratio, max_steps, log_start=-2, log_base=10
 75    ):
 76        """
 77        Generates a decay schedule for a given initial value.
 78
 79        Parameters
 80        ----------
 81        init_value : float
 82            Finite initial value of the quantity being decayed.
 83        min_value : float
 84            Finite minimum value, no greater than `init_value`.
 85        decay_ratio : float
 86            Fraction of `max_steps` over which values decay, in (0, 1].
 87        max_steps : int
 88            Positive number of steps in the returned schedule.
 89        log_start : float, optional
 90            Finite negative start of the logarithmic sequence, by default -2.
 91        log_base : float, optional
 92            Finite logarithm base greater than 1, by default 10.
 93
 94        Returns
 95        -------
 96        np.ndarray
 97            Decay values where values[i] is the value used at i-th step.
 98        """
 99        if (
100            isinstance(max_steps, bool)
101            or not isinstance(max_steps, Integral)
102            or max_steps < 1
103        ):
104            raise ValueError("max_steps must be a positive integer")
105        if (
106            isinstance(decay_ratio, bool)
107            or not isinstance(decay_ratio, Real)
108            or not np.isfinite(decay_ratio)
109            or not 0 < decay_ratio <= 1
110        ):
111            raise ValueError("decay_ratio must be finite and in (0, 1]")
112        if any(
113            isinstance(value, bool)
114            or not isinstance(value, Real)
115            or not np.isfinite(value)
116            for value in (init_value, min_value)
117        ):
118            raise ValueError("init_value and min_value must be finite")
119        if init_value < min_value:
120            raise ValueError("init_value must be greater than or equal to min_value")
121        if (
122            isinstance(log_start, bool)
123            or not isinstance(log_start, Real)
124            or not np.isfinite(log_start)
125            or log_start >= 0
126        ):
127            raise ValueError("log_start must be finite and negative")
128        if (
129            isinstance(log_base, bool)
130            or not isinstance(log_base, Real)
131            or not np.isfinite(log_base)
132            or log_base <= 1
133        ):
134            raise ValueError("log_base must be finite and greater than 1")
135
136        max_steps = int(max_steps)
137        if max_steps == 1 or init_value == min_value:
138            return np.full(max_steps, init_value, dtype=float)
139
140        decay_steps = min(max_steps, max(2, int(max_steps * decay_ratio)))
141        rem_steps = max_steps - decay_steps
142        values = np.logspace(log_start, 0, decay_steps, base=log_base, endpoint=True)[
143            ::-1
144        ]
145        values = (values - values.min()) / (values.max() - values.min())
146        values = (init_value - min_value) * values + min_value
147        values = np.pad(values, (0, rem_steps), constant_values=min_value)
148        return values
